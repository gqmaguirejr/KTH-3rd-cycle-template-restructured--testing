% This file contains the LaTeX to add information to the PDF file (specifically, author(s), title(s), and keywords
% It uses the hyperref package and should be included before the \begin{document}
%
% I want to acknowledge the inspiration of Karl Voit's template for TU Graz that inspired me to add the PDF document information
% For more information about his template see https://github.com/novoid/LaTeX-KOMA-template
% Note that this template does not use anything from his template other than the names of the information for the PDF meta fields, i.e., mytitle, myauthor, and mykeywords together with the idea of defining the corresponding newcommand to set the relevant hyperref parameters.

\makeatletter
\ifx\@subtitle\@empty
    \newcommand{\mytitle}{\@title}
\else
    \ifinswedish
        \newcommand{\mytitle}{\@title\xspace–\xspace\@subtitle}
    \else
        \newcommand{\mytitle}{\@title: \@subtitle}
    \fi
\fi
\makeatother

% Put the alternative title (and subtitle) into the PDF Subject meta
\makeatletter
\ifx\@altsubtitle\@empty\relax
    \newcommand{\myalttitle}{\@alttitle}
\else
    \ifinswedish
        \newcommand{\myalttitle}{\@alttitle: \@altsubtitle}
    \else
    \newcommand{\myalttitle}{\@alttitle\xspace–\xspace\@altsubtitle}
    \fi
    
\fi
\makeatother

\ifinswedish
\hypersetup{%
    pdflang={sv},
    pdfmetalang={sv},
    %pdftitle={\mytitle}        % Title field
    pdftitle = {[sv]{{\mytitle}}, [en]{{\myalttitle}}},
}
\else
\hypersetup{%
    pdflang={en},
    pdfmetalang={en},
    %pdftitle={\mytitle}        % Title field
    pdftitle = {[en]{{\mytitle}}, [sv]{{\myalttitle}}},
}
\fi

\makeatletter

\ifx\@secondAuthorsLastname\@empty
    \newcommand{\myauthor}{\@authorsFirstname\space\@authorsLastname} 
\else
    \newcommand{\myauthor}{\@authorsFirstname\space\@authorsLastname,
\space\@secondAuthorsFirstname\space\@secondAuthorsLastname}
\fi

\makeatother

\hypersetup{
     pdfauthor={\myauthor}      % Author field
}


\ExplSyntaxOn
\makeatletter
\cs_new_protected:Npn \build_tagged_keywords: {
  \clist_clear:N \l_tmpa_clist

  % Add English keywords tagged with [en]
  \tl_if_empty:NF \@EnglishKeywords {
    \clist_map_inline:Nn \@EnglishKeywords {
      \tl_set:Nn \l_tmpa_tl {##1}
      \tl_trim_spaces:N \l_tmpa_tl
      \clist_put_right:Ne \l_tmpa_clist { [en]{\exp_not:V \l_tmpa_tl} }
    }
  }

  % Add Swedish keywords tagged with [sv]
  \tl_if_empty:NF \@SwedishKeywords {
    \clist_map_inline:Nn \@SwedishKeywords {
      \tl_set:Nn \l_tmpa_tl {##1}
      \tl_trim_spaces:N \l_tmpa_tl
      \clist_put_right:Ne \l_tmpa_clist { [sv]{\exp_not:V \l_tmpa_tl} }
    }
  }

  % Construct the complete argument for \hypersetup: "pdfkeywords = { ... }"
  \tl_set:Ne \l_tmpb_tl { pdfkeywords = { \clist_use:Nn \l_tmpa_clist { ,~ } } }

  % Pass the single formatted key-value pair to \hypersetup
  \exp_args:NV \hypersetup \l_tmpb_tl
}

\cs_set_eq:NN \buildTaggedKeywords \build_tagged_keywords:
\makeatother
\ExplSyntaxOff

\cs_new_protected:Npn \build_pdf_subject: {
  % 1. Determine localized degree string with XXX placeholder handling
  \ifinswedish
    \str_case:VnF \@degreeName {
      { Doctorate }  { \tl_set:Nn \l_tmpa_tl { Doktorsavhandling } }
      { Licentiate } { \tl_set:Nn \l_tmpa_tl { Licentiatavhandling } }
      { XXX }        { \tl_set:Nn \l_tmpa_tl { XXX-avhandling } }
    } { \tl_set:Nn \l_tmpa_tl { Avhandling } }
    \tl_set:Ne \l_tmpb_tl { \l_tmpa_tl \space inom \space \exp_not:V \@subjectArea }
  \else
    \str_case:VnF \@degreeName {
      { Doctorate }  { \tl_set:Nn \l_tmpa_tl { Doctoral~Thesis } }
      { Licentiate } { \tl_set:Nn \l_tmpa_tl { Licentiate~Thesis } }
      { XXX }        { \tl_set:Nn \l_tmpa_tl { XXX~Thesis } }
    } { \tl_set:Nn \l_tmpa_tl { Thesis } }
    \tl_set:Ne \l_tmpb_tl { \l_tmpa_tl \space in \space \exp_not:V \@subjectArea }
  \fi

  % 2. Pack as a single key-value token list
  \tl_set:Ne \l_tmpa_tl { pdfsubject = { \exp_not:V \l_tmpb_tl } }

  % 3. Pass the clean key-value pair to \hypersetup
  \exp_args:NV \hypersetup \l_tmpa_tl
}

% Add the TRITA number to the metadata
% Get and store information about the series and the number within this series, i.e, TRITA numbers
%"Series": \{
%	"Title of series": "TRITA-ICT-EX"
%	"No. in series": "2019:00"
\ExplSyntaxOn
\makeatletter
\cs_generate_variant:Nn \pdfmeta_xmp_add:n { e }

\cs_new_protected:Npn \inject_prism_metadata: {
  \tl_clear:N \l_tmpa_tl

  % Series / Volume
  \tl_if_empty:NF \@thesisSeries {
    \tl_put_right:Ne \l_tmpa_tl {
      \space\space\space\space<prism:volume>\exp_not:V \@thesisSeries</prism:volume>^^J
    }
  }

  % Issue / Series Number
  \tl_if_empty:NF \@thesisSeriesNumber {
    \tl_put_right:Ne \l_tmpa_tl {
      \space\space\space\space<prism:number>\exp_not:V \@thesisSeriesNumber</prism:number>^^J
    }
  }

  % Publication Type (already registered as 'aggregationType' or custom property)
  \tl_put_right:Nn \l_tmpa_tl {
    \space\space\space\space<prism:publicationType>report</prism:publicationType>^^J
  }

  % Inject properties directly into the existing outer <rdf:Description>
  \pdfmeta_xmp_add:e { \l_tmpa_tl }
}

\cs_set_eq:NN \injectPrismMetadata \inject_prism_metadata:
\makeatother
\ExplSyntaxOff

\ExplSyntaxOn
\makeatletter
\cs_new_protected:Npn \inject_publisher_metadata: {
  \tl_clear:N \l_tmpa_tl

  \ifinswedish
    \tl_set:Nn \l_tmpa_tl { Kungliga~Tekniska~högskolan~(KTH) }
  \else
    \tl_set:Nn \l_tmpa_tl { KTH~Royal~Institute~of~Technology }
  \fi

  % Inject directly into dc:publisher as an rdf:Bag
  \pdfmeta_xmp_add:e {
    \space\space\space\space<dc:publisher>^^J
    \space\space\space\space\space\space<rdf:Bag>^^J
    \space\space\space\space\space\space\space\space<rdf:li>\exp_not:V \l_tmpa_tl</rdf:li>^^J
    \space\space\space\space\space\space</rdf:Bag>^^J
    \space\space\space\space</dc:publisher>^^J
  }
}
