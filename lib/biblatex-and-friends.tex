% Source - https://tex.stackexchange.com/a/163787
% Posted by moewe, modified by community. See post 'Timeline' for change history
% Retrieved 2026-06-11, License - CC BY-SA 3.0
% Augmented by GQMJr to add second mapping and comments for both

\DeclareSourcemap{
  \maps[datatype=bibtex]{
    % Map 1: Aliasing pmid to pubmed
    \map{
      \step[fieldsource=pmid, fieldtarget=pubmed]
      }
    % Map 2: Aliasing mrnumber to mr
    \map{
      \step[fieldsource=mrnumber, fieldtarget=mr]
    }
    % Map 3 and 4: Conditionally strip English language indicators
    \map{
      % Check if 'language' field matches "English" or "eng" (case-insensitive)
      % note that one has to use '\A' rather than '^' and '\z' rather than a '$' due to Overleaf's parser
      % The 'final' directive causes an early termination of the processing, hence it does not execute the following step.
      \step[fieldsource=language, match=\regexp{\A(English|eng)\z}, final]
      \step[fieldset=language, null]
    }
    \map{
      % Check if 'langid' field matches "English" or "eng" (case-insensitive)
      \step[fieldsource=langid, match=\regexp{\A(English|eng)\z}, final]
      \step[fieldset=langid, null]
    }
  }
}

%% Style Portability:
% Some patches rely on overriding low-level macros (such as volume+number+eid and \xpatchbibdriver{inproceedings}) specifically tailored to style=ieee. These hooks may need slight adjustments if switching to authoryear or chicago-authordate formats.

% fix the way that volume and number are output
% 1. Strip the default comma prefix that styles inject before numbers
% Declare the structural format for the 'number' field specifically within this macro
\DeclareFieldFormat[article,periodical]{number}{\mkbibparens{#1}}

% 2. Intercept and clear the contextual punctuation buffer specifically for the journal volume string
\renewbibmacro*{journal+issuetitle}{%
  \usebibmacro{journal}%
  \setunit*{\addspace}%
  \iffieldundef{series}
    {}
    {\newunit
     \printfield{series}%
     \setunit*{\addspace}}%
  \usebibmacro{volume+number+eid}%
}

% 3. Force the absolute execution override to eliminate the nested separator
\renewbibmacro*{volume+number+eid}{%
  \printfield{volume}%
  \unspace% <--- FORCE-STRIP ANY COMMAS CURRENTLY SITTING IN THE PIPELINE
  \printfield{number}%
  \setunit{\addcomma\space}%
  \printfield{eid}%
}



% 1. Inject the master ISSN printing block into the @inproceedings (Conference) driver
\xpatchbibdriver{inproceedings}

  {\printfield{issn}\newunit\newblock\usebibmacro{doi+eprint+url}}
